cmake_minimum_required(VERSION 3.27)
project(MEFSingleLineBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LESDK_WITH_LE1 ON CACHE BOOL "" FORCE)
set(LESDK_WITH_LE2 OFF CACHE BOOL "" FORCE)
set(LESDK_WITH_LE3 OFF CACHE BOOL "" FORCE)
set(LESDK_WITH_LEL OFF CACHE BOOL "" FORCE)
set(LESDK_WITH_SAMPLES OFF CACHE BOOL "" FORCE)
set(LESDK_WITH_TESTS OFF CACHE BOOL "" FORCE)

# Dependency stays upstream; do not redistribute the generated UE3 SDK here.
set(LESDK_SOURCE_DIR "" CACHE PATH "Optional existing LExSDKv2 checkout")
if(NOT LESDK_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(lexsdk
        GIT_REPOSITORY https://github.com/llamathings/LExSDKv2.git
        GIT_TAG 3885be14113816ff1d2a0d941aa96bba6df9dc58
        GIT_PROGRESS TRUE)
    FetchContent_MakeAvailable(lexsdk)
else()
    add_subdirectory("${LESDK_SOURCE_DIR}" sdk)
endif()
add_library(MEFSingleLineBridge SHARED Bridge.cpp)
target_link_libraries(MEFSingleLineBridge PRIVATE LESDK-LE1 bcrypt)
target_compile_definitions(MEFSingleLineBridge PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(MEFSingleLineBridge PRIVATE /W4 /utf-8 /bigobj /permissive /wd4121)
set_target_properties(MEFSingleLineBridge PROPERTIES SUFFIX ".asi")
enable_testing()
add_executable(BridgeNativeTests NativeTests.cpp)
add_test(NAME BridgeNativeTests COMMAND BridgeNativeTests)
add_executable(BridgeRampTests RampTests.cpp)
add_test(NAME BridgeRampTests COMMAND BridgeRampTests)
add_executable(BridgeBindingTests BindingTests.cpp)
add_test(NAME BridgeBindingTests COMMAND BridgeBindingTests "${CMAKE_CURRENT_LIST_DIR}/../../../tests/data/binding.log")
add_executable(BridgeLoaderTests LoaderTests.cpp)
add_dependencies(BridgeLoaderTests MEFSingleLineBridge)
add_test(NAME BridgeLoaderTests COMMAND BridgeLoaderTests "$<TARGET_FILE:MEFSingleLineBridge>")
add_executable(BridgeMaterialProbeTests MaterialProbeTests.cpp)
add_test(NAME BridgeMaterialProbeTests COMMAND BridgeMaterialProbeTests)
add_executable(BridgeRenderTraceTests RenderTraceTests.cpp)
add_test(NAME BridgeRenderTraceTests COMMAND BridgeRenderTraceTests)
add_executable(BridgeResourceTraceTests ResourceTraceTests.cpp)
add_test(NAME BridgeResourceTraceTests COMMAND BridgeResourceTraceTests)
add_executable(BridgeFacialClipTests FacialClipTests.cpp)
add_test(NAME BridgeFacialClipTests COMMAND BridgeFacialClipTests "${CMAKE_CURRENT_LIST_DIR}/../../../tests/data/clips")
add_executable(BridgeFaceOwnershipTests FaceOwnershipTests.cpp)
add_test(NAME BridgeFaceOwnershipTests COMMAND BridgeFaceOwnershipTests)
add_executable(BridgeFaceRoutingTests FaceRoutingTests.cpp)
add_test(NAME BridgeFaceRoutingTests COMMAND BridgeFaceRoutingTests)

add_executable(BridgeFaceTransitionTests FaceTransitionTests.cpp)
add_test(NAME BridgeFaceTransitionTests COMMAND BridgeFaceTransitionTests)
